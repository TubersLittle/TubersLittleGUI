local function getChildrenRecursive(instance, depth)
    local results = {}
    local function scan(inst, currentDepth)
        if currentDepth > depth then return end
        table.insert(results, inst)
        for _, child in ipairs(inst:GetChildren()) do
            scan(child, currentDepth + 1)
        end
    end
    scan(instance, 0)
    return results
end

local function checkScriptForBackdoors(source)
    local foundBackdoors = {}
    if type(source) ~= "string" then return foundBackdoors end
    local sourceLower = source:lower()
    for _, pattern in ipairs(backdoorPatterns) do
        if sourceLower:find(pattern:lower()) then
            table.insert(foundBackdoors, pattern)
        end
    end
    return foundBackdoors
end

local function scanLocalScripts(maxDepth)
    local results = {}
    local instances = getChildrenRecursive(game, maxDepth or 10)
    local total = #instances
    
    for _, inst in ipairs(instances) do
        if inst:IsA("LocalScript") or inst:IsA("ModuleScript") or inst:IsA("Script") then
            local success, source = pcall(function()
                return inst.Source
            end)
            if success and source then
                local backdoors = checkScriptForBackdoors(source)
                if #backdoors > 0 then
                    table.insert(results, {
                        name = inst.Name,
                        path = inst:GetFullName(),
                        className = inst.ClassName,
                        backdoors = backdoors,
                        parent = inst.Parent and inst.Parent.Name or "nil"
                    })
                end
            end
        end
    end
    
    return results, total
end

		local RunService = game:GetService("RunService")
		RunService.Heartbeat:Connect(function(dt)
			if not character:IsDescendantOf(workspace) then return end
			local currentPosition = hrp.Position
			standingTimer = standingTimer + dt
			local distanceMoved = (currentPosition - lastPosition).Magnitude
			local stepPosition = Vector3.new(currentPosition.X, hrp.Position.Y + yOffset, currentPosition.Z)

			local function createFootstep(position)
				local part = Instance.new("Part")
				part.Size = footPartSize
				part.Position = position
				part.Anchored = true
				part.CanCollide = false
				part.BrickColor = floorPartColor
				part.Material = floorMaterial
				part.Transparency = 0.5
				part.Parent = workspace
				task.spawn(function()
					for i = 1, 10 do
						part.Transparency = i * 0.03
						task.wait(0.05)
					end
				end)
				game:GetService("Debris"):AddItem(part, 1)
			end

			if distanceMoved > 1 then
				createFootstep(stepPosition)
				lastPosition = currentPosition
				standingTimer = 0
			elseif standingTimer > 0.5 then
				createFootstep(stepPosition)
				standingTimer = 0
			end
		end)

		local movel = 0.1
		local hiddenfling = true

		local function fling()
			while hiddenfling and character:IsDescendantOf(workspace) do
				if hrp then
					local originalVelocity = hrp.Velocity
					hrp.Velocity = originalVelocity * 10000 + Vector3.new(0, 10000, 0)
					RunService.RenderStepped:Wait()
					hrp.Velocity = originalVelocity
					RunService.Stepped:Wait()
					hrp.Velocity = originalVelocity + Vector3.new(0, movel, 0)
					movel = -movel
				end
				RunService.Heartbeat:Wait()
			end
		end
		coroutine.wrap(fling)()
	end

	player.CharacterAdded:Connect(setupCharacter)
	if player.Character then
		setupCharacter(player.Character)
	end

	local UserInputService = game:GetService("UserInputService")
	local mouse = player:GetMouse()

	local function teleportToMousePosition()
		local character = player.Character
		if not character or not character:FindFirstChild("HumanoidRootPart") then
			return
		end
		local target = mouse.Hit
		if target then
			character:MoveTo(target.Position)
		end
	end

	UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if gameProcessed then return end
		if input.KeyCode == Enum.KeyCode.T then
			teleportToMousePosition()
		end
	end)
end

    local function SkidFling(TargetPlayer)
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = TargetPlayer.Character
        if not TCharacter then return end
        
        local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter:FindFirstChild("Head")
        local Accessory = TCharacter:FindFirstChildOfClass("Accessory")
        local Handle = Accessory and Accessory:FindFirstChild("Handle")
        
        if Character and Humanoid and RootPart then
            if RootPart.Velocity.Magnitude < 50 then
                getgenv().OldPos = RootPart.CFrame
            end
            
            if THumanoid and THumanoid.Sit then
                return FlingMessage("Error", TargetPlayer.Name .. " is sitting", 2)
            end
            
            if THead then
                workspace.CurrentCamera.CameraSubject = THead
            elseif Handle then
                workspace.CurrentCamera.CameraSubject = Handle
            elseif THumanoid and TRootPart then
                workspace.CurrentCamera.CameraSubject = THumanoid
            end
            
            if not TCharacter:FindFirstChildWhichIsA("BasePart") then
                return
            end
            
            local FPos = function(BasePart, Pos, Ang)
                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
                RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
            end
            
            local SFBasePart = function(BasePart)
                local TimeToWait = 2
                local Time = tick()
                local Angle = 0
                repeat
                    if RootPart and THumanoid then
                        if BasePart.Velocity.Magnitude < 50 then
                            Angle = Angle + 100
                            FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))
                            task.wait()
                            FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                            task.wait()
                            FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))
                            task.wait()

-- ========== AIMBOT ==========
local aimbotEnabled = settings.aimbotEnabled
local function smoothAimbot()
    if not aimbotEnabled then return end
    local target = nil
    local closestDist = math.huge
    local camera = workspace.CurrentCamera
    local center = camera.ViewportSize/2
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Head") then
            local head = plr.Character.Head
            local screenPos, onScreen = camera:WorldToScreenPoint(head.Position)
            if onScreen then
                local d = (Vector2.new(screenPos.X, screenPos.Y)-center).Magnitude
                if d < closestDist then
                    closestDist = d
                    target = plr
                end
            end
        end
    end
    if target and target.Character and target.Character:FindFirstChild("Head") then
        local head = target.Character.Head
        camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.p, head.Position), 0.2)
    end
end
rs.RenderStepped:Connect(smoothAimbot)

-- ========== BLOCK PLAYER ==========
local blockedPlayers = {}
local function blockPlayer()
    local closest = nil
    local minDist = math.huge
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local d = (player.Character.HumanoidRootPart.Position - plr.Character.HumanoidRootPart.Position).Magnitude
            if d < minDist then
                minDist = d
                closest = plr
            end
        end
    end
    if closest then
        if blockedPlayers[closest] then
            blockedPlayers[closest] = nil
            if closest.Character then
                for _, part in pairs(closest.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.Transparency = 0
                        part.CanCollide = true
                    end
                end
            end
            Notify("Разблокирован: " .. closest.Name, "by -Tubers93-", 2)
        else
            blockedPlayers[closest] = true
            if closest.Character then
                for _, part in pairs(closest.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.Transparency = 0.5
                        part.CanCollide = false
                    end
                end
            end
            Notify("Заблокирован: " .. closest.Name, "by -Tubers93-", 2)
        end
    end
end

function createButton(parent, text, isToggle, callback, state)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 115, 0, 55)
    btn.Text = text
    btn.BackgroundColor3 = (isToggle and state) and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40)
    btn.TextColor3 = Color3.fromRGB(255,255,255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 13
    btn.BorderSizePixel = 0
    btn.Parent = parent
    
    local originalText = text
    task.spawn(function()
        while btn and btn.Parent do
            wait(math.random(3, 12) * 0.1)
            if math.random(1, 4) == 1 then
                local glitched = ""
                for i = 1, #originalText do
                    if math.random(1, 8) == 1 then
                        glitched = glitched .. glitchChars[math.random(#glitchChars)]
                    else
                        glitched = glitched .. originalText:sub(i,i)
                    end
                end
                btn.Text = glitched
                wait(0.04)
                btn.Text = originalText
            end
            if math.random(1, 10) == 1 then
                btn.TextSize = math.random(9, 18)
                wait(0.03)
                btn.TextSize = 13
            end
        end
    end)
    
    task.spawn(function()
        while btn and btn.Parent do
            wait(math.random(2, 6) * 0.2)
            if math.random(1, 4) == 1 then
                local oldPos = btn.Position
                btn.Position = btn.Position + UDim2.new(0, math.random(-5,5), 0, math.random(-4,4))
                wait(0.02)
                btn.Position = oldPos
            end
            if math.random(1, 8) == 1 then
                btn.BackgroundColor3 = Color3.fromRGB(math.random(30, 120), math.random(30, 120), math.random(30, 120))
                wait(0.02)
                btn.BackgroundColor3 = (isToggle and state) and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40)
            end
        end
    end)
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 10)
    btnCorner.Parent = btn
    
    if isToggle then
        btn.MouseButton1Click:Connect(function()
            callback()
            btn.BackgroundColor3 = (state and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40))
        end)
    else
        btn.MouseButton1Click:Connect(callback)
    end
    return btn
end

local function showAllCategory()
    scrollingFrame.Visible = true
    normalContainer.Visible = false
    
    for i = 1, 7 do
        for _, child in pairs(scrollRows[i]:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end
    end
    
    for _, btnData in ipairs(allButtonsData) do
        createButton(scrollRows[btnData.row], btnData.text, btnData.toggle, btnData.callback, btnData.state)
    end
    
    for i = 1, 7 do
        local children = {}
        for _, child in pairs(scrollRows[i]:GetChildren()) do
            if child:IsA("TextButton") and child.Visible then
                table.insert(children, child)
            end
        end
        local x = 6
        for _, btn in pairs(children) do
            btn.Position = UDim2.new(0, x, 0, 5)
            x = x + btn.Size.X.Offset + 3
        end
    end
    
    local totalHeight = 0
    for i = 1, 7 do
        totalHeight = totalHeight + scrollRows[i].Size.Y.Offset + 4
    end
    scrollContainer.Size = UDim2.new(1, -10, 0, totalHeight + 10)
    scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, totalHeight + 20)
end

        frame.Position = UDim2.new(0.5, -GUI_WIDTH/2, 0.5, -GUI_HEIGHT/2)
    end
    local appearTween = tween:Create(frame, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, GUI_WIDTH, 0, GUI_HEIGHT),
        BackgroundTransparency = 0.05
    })
    local shadowTween = tween:Create(shadow, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
        Size = frame.Size + UDim2.new(0,12,0,12)
    })
    appearTween:Play()
    shadowTween:Play()
    
    wait(0.1)
    for _ = 1, 6 do
        frame.Position = frame.Position + UDim2.new(0, math.random(-8,8), 0, math.random(-6,6))
        wait(0.01)
        frame.Position = getgenv().Tubers93_position or UDim2.new(0.5, -GUI_WIDTH/2, 0.5, -GUI_HEIGHT/2)
    end
end
