-- BLOCK PLAYER
local blockedPlayers = {}
local function blockPlayer()
    local closest = nil
    local minDist = math.huge
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local d = (player.Character.HumanoidRootPart.Position - plr.Character.HumanoidRootPart.Position).Magnitude
            if d < minDist then
                minDist = d
                closest = plr
            end
        end
    end
    if closest then
        if blockedPlayers[closest] then
            blockedPlayers[closest] = nil
            if closest.Character then
                for _, part in pairs(closest.Character:GetDescendants()) do
                    if part:IsA("BasePart") then

            blockedPlayers[closest] = true

                for _, part in pairs(closest.Character:GetDescendants()) do

                        part.Transparency = 0.5

    end

function createButton(parent, text, isToggle, callback, state)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 115, 0, 55)
    btn.Text = text
    btn.BackgroundColor3 = (isToggle and state) and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40)
    btn.TextColor3 = Color3.fromRGB(255,255,255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 13
    btn.BorderSizePixel = 0
    btn.Parent = parent
    
    -- БЫСТРЫЙ глюк текста кнопки (каждые 0.3-2 сек)
    local originalText = text
    task.spawn(function()
        while btn and btn.Parent do
            wait(math.random(3, 12) * 0.1)
            if math.random(1, 4) == 1 then
                local glitched = ""
                for i = 1, #originalText do
                    if math.random(1, 8) == 1 then
                        glitched = glitched .. glitchChars[math.random(#glitchChars)]
                    else
                        glitched = glitched .. originalText:sub(i,i)
                    end
                end
                btn.Text = glitched
                wait(0.04)
                btn.Text = originalText
            end
            if math.random(1, 10) == 1 then
                btn.TextSize = math.random(9, 18)
                wait(0.03)
                btn.TextSize = 13
            end
            if math.random(1, 12) == 1 then
                btn.TextColor3 = Color3.fromRGB(math.random(200,255), math.random(200,255), math.random(200,255))
                wait(0.03)
                btn.TextColor3 = Color3.fromRGB(255,255,255)
            end
        end
    end)
    
    -- БЫСТРЫЙ глюк: кнопки дёргаются чаще
    task.spawn(function()
        while btn and btn.Parent do
            wait(math.random(2, 6) * 0.2)
            if math.random(1, 4) == 1 then
                local oldPos = btn.Position
                btn.Position = btn.Position + UDim2.new(0, math.random(-5,5), 0, math.random(-4,4))
                wait(0.02)
                btn.Position = oldPos
            end
            if math.random(1, 8) == 1 then
                btn.BackgroundColor3 = Color3.fromRGB(math.random(30, 120), math.random(30, 120), math.random(30, 120))
                wait(0.02)
                btn.BackgroundColor3 = (isToggle and state) and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40)
            end
        end
    end)
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 10)
    btnCorner.Parent = btn
    if isToggle then
        btn.MouseButton1Click:Connect(function()
            callback()
            btn.BackgroundColor3 = (state and Color3.fromRGB(80,80,80) or Color3.fromRGB(40,40,40))
        end)
    else
        btn.MouseButton1Click:Connect(callback)
    end
    return btn
end
